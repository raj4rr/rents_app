How Thread is Death in Java Program
Previous Home Next

Every running thread should be dead in two way:

  1. if run() is completely execution then thread is dies naturally.
  2. by calling method like stop() and dead(), these method is used to dies the thread manually. 
//following example for dead method
package r4r.co.in;
class MYThread implements Runnable {
String name;    // name of thread
Thread t;
MYThread(String threadname) {
name = threadname;
t = new Thread(this, name);
System.out.println("My thread: " + t);
t.start();   // Thread Start for execution
}
public void run() {  // Initialization of  thread.
try {
for (int i = 0; i < 5; i++) {
System.out.println(name + ": " + i);
t.stop();   //kill the thread as initilized
}
} catch (Exception e) {
System.out.println(name + e);
}
}
}
class ThreadDemo {
public static void main(String args[]) {
new MYThread("One");
}
}
Notify, NotifyAll and Wait in Java Programming

Example

// A implementation of a wait and notify in producer and consumer.
package r4r.co.in;
class Q { 
int n;
boolean valueSet = false;
synchronized int get() {
if (!valueSet) {
try {
wait();
} catch (InterruptedException e) {
System.out.println("Exception" + e);
}
}
System.out.println("Consumer: " + n);
valueSet = false;
notify();
return n;
}
synchronized void put(int n) {
if (valueSet) {
try {
wait();
} catch (InterruptedException e) {
System.out.println("Exception" + e);
}
}
this.n = n;
valueSet = true;
System.out.println("Producer: " + n);
notify();
}
}
class Producer implements Runnable {
Q q;
Producer(Q q) {
this.q = q;
new Thread(this, "Producer").start();
}
public void run() {
int i = 1;
while (i < 20) {
q.put(i++);
}
}
}
class Consumer implements Runnable {
Q q;
Consumer(Q q) {
this.q = q;
new Thread(this, "Consumer").start();
}
public void run() {
while (true) {
q.get();
}
}
}
class PCFixed {
public static void main(String args[]) {
Q q = new Q();
new Producer(q);
new Consumer(q);
}
}
