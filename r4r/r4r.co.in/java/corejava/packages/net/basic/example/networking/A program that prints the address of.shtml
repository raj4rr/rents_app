The Program using Internet Address


import java.net.*;

class r4r {

  public static void main (String args[]) {

    try {
      InetAddress address = InetAddress.getByName("www.r4r.co.in");
      System.out.println(address);
    }
    catch (UnknownHostException e) {
      System.out.println("Could not find www.r4r.co.in");
    }
}
}
