Array reflection is use whether an argument or object is an array, and easily retrieve the Object class. Let consider the following program-

Example

/*
 * Save as a DoubleArray.java
 * Program for using reflection to create, fill, and display an array.
 */
package r4r.co.in;

import java.lang.reflect.Array;
import java.util.Random;

public class DoubleArray {

public static void main(String[] args) {
//Create an Array, and define the Size 
Object array = Array.newInstance(int.class, 5);
//Three different method define
printArray(array);   
fillArray(array);
displayArray(array);
}
private static void printArray(Object array) {
Class class1 = array.getClass();
if (class1.isArray()) {
Class element1 = class1.getComponentType();
System.out.println("Type of Array:: "
+ element1 + "
              ,Element in Array::" + Array.getLength(array));
}
}
private static void fillArray(Object array) {
int fiLength = Array.getLength(array);
Random generator = new Random(System.nanoTime());
for (int i = 0; i < fiLength; i++) {
int random = generator.nextInt();
Array.setInt(array, i,random);
}
}
private static void displayArray(Object array) {
int disLength = Array.getLength(array);
for (int i = 0; i < disLength; i++) {
int value = Array.getInt(array, i);
System.out.println(" At position: " + i + ", value of Element: " + value);
}
}
}
