A program for Map Interface

/*
 * Save as a MapExample.java
 * Program for Map interface
 */
package r4r.co.in;

import java.util.*;

public class MapExample {

public static void main(String[] args) {

//Create a HashMap, extends AbstractMap
HashMap hashMap = new HashMap();
// Put elements to the map
hashMap.put(new Integer(1), "A");
hashMap.put(new Integer(2), "B");
hashMap.put(new Integer(3), "C");
hashMap.put(new Integer(4), "D");
hashMap.put(new Integer(5), "E");
hashMap.put(new Integer(6), "F");

// Get a set of the entries
Set set = hashMap.entrySet();

// Get an iterator
Iterator itr = set.iterator();
while (itr.hasNext()) {
Map.Entry entry = (Map.Entry) itr.next();
int key = (int) entry.getKey();
String value = (String) entry.getValue();
//int hashcode= entry.hashCode();
System.out.println("Key :" + key + "  Value :" + value);
}
}
}
