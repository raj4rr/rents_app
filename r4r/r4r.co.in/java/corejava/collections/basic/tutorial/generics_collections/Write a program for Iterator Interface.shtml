A program for Iterator Interface

/*
 * Save as a IteratorExample.java
 * Program for Using Iterator Interface
 */
package r4r.co.in;
import java.lang.reflect.Array;
import java.util.Iterator;
class IteratorExample implements Iterator<Integer> {
private int count;
private int index;
private Object array;
private IteratorExample(Object[] object) {
count=Array.getLength(object);
array=object;
}  

//    @Override
public Iterator<Integer> iterator() {
return this ;
}
@Override
public boolean hasNext() {
return (index < count);
}

@Override
public Integer next() {
return (Integer) Array.get(array, index++);
}

@Override
public void remove() {

index--;
}

public static void main(String[] args) {

Object object[]=new Object[]{1, 2, 3, 4, 5};

Iterator iterator= (Iterator) new IteratorExample(object);
while (iterator.hasNext()) {
System.out.println("Element are::" +iterator.next());
}       
}
}
