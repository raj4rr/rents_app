A program for Using ListIterator

/*
 * Save as a IteratorExaple.java
 * Program for Using ListIterator Inerface
 */
package r4r.co.in;
import java.util.ArrayList;
import java.util.ListIterator;
public class IteratorExaple {
public static void main(String[] args) {
//Create a ArrayList
ArrayList arrayList = new ArrayList();
//Add some Element into ArrayList
arrayList.add("A");
arrayList.add("B");
arrayList.add("C");
arrayList.add("D");
arrayList.add("E");
arrayList.add("F");

System.out.println("Size::" + arrayList.size() + " and,Element in AarrayList:: " + arrayList);

//Print Element By IteratorList Interface(Member of Collection FrameWork)
System.out.println("Element from the ArrayList::");
ListIterator listIterator = arrayList.listIterator();
//Now Check everyElement into the List
while (listIterator.hasNext()) {
Object object = listIterator.next();
System.out.print(object + "  ");
}
System.out.print("\nPrint the Character into reverse order:: ");
//Read List in reverse order
while (listIterator.hasPrevious()) {
Object object = listIterator.previous();
System.out.print( object + "  ");
}
}
}

output

Result::

Element from the ArrayList::
A  B  C  D  E  F 
Print the Character into reverse order:: F  E  D  C  B  A
