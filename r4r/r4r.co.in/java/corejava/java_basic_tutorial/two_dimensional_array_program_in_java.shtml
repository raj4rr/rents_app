//write a program to short a matrix  in  Rowise 

import java.util.Scanner;
class Row
{
	public static void main(String args[])
	{
		Scanner sc;
		sc=new Scanner(System.in);
		System.out.print("Enter how many rows ");
		int r=sc.nextInt();
		System.out.print("Enter how many cols ");
		int c=sc.nextInt();
		int m[][];
		m=new int[r][c];

		/*input*/
		for(int i=0;i<r;i++)
			for(int j=0;j<c;j++)
			{
				System.out.print("Enter element "+(i+1)+" "+(j+1)+" ");
				m[i][j]=sc.nextInt();
			}

		/*sorting logic*/
		for(int k=0;k<r;k++)
			for(int i=0;i<c-1;i++)
				for(int j=i+1;j<c;j++)
				{
					if(m[k][i]>m[k][j])
					{
						int t=m[k][i];
						m[k][i]=m[k][j];
						m[k][j]=t;
					}
				}

		/*output*/
		for(int i=0;i<r;i++)
		{
			for(int j=0;j<c;j++)
				System.out.print(m[i][j]+"\t");
				System.out.println();
		}

		}		
}
