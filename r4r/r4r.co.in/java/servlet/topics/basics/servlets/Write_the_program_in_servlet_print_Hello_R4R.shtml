Following program of print Hello R4R in the servlet.

Save the following file as HelloR4R1.java

package R4R.co.in;
import java.io.*;
import javax.servlet.*;
import javax.servlet.annotation.WebServlet; 
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet
 (name = "HelloR4R1", urlPatterns = {"/HelloR4R1"})

 public class HelloR4R1 extends HttpServlet {

 public void service
 (HttpServletRequest request, HttpServletResponse response)
   throws ServletException, IOException {
response.setContentType("text/html;charset=UTF-8");
  PrintWriter out = response.getWriter();
  try {
       /* TODO output your page here*/
       out.println("<html>"); 
       out.println("<head>");
       out.println("<title> HelloR4R</title>");
       out.println("</head>");
       out.println("<body>");
       out.println("<h1>FirstServlet Example: HelloR4R </h1>");
       out.println("</body>");
       out.println("</html>");

   }
   finally
  {
   out.close();
  }

}

}
