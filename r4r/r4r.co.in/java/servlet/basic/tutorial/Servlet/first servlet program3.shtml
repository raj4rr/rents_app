First Servlet Program:

In order to compile a servlet, jar file containing servlet API classes and interfaces need to be available in the classpath.Servlet API classes and interfaces are not provided by sun microsystem as part of standard java library.These are provided by webserver and application vendors i.e; in order to compile servlets ,the web server and the application server need to be installed.

index.html

<html>
	<head>
		<title>First web application</title>
	</head>
	<body>
		<form action="get" method="firstServlet">
			Name<input type="text" name="txtName">
			<input type="submit" value="submit">
		</form>
	</body>
</html>

web.xml

<web-app>
	<servlet>
		<servlet-name>g1</servlet-name>
		<servlet-class>FirstServlet</servlet-class>
	</servlet>
	<servlet-mapping>
		<servlet-name>g1</servlet-name>
		<url-pattern>firstServlet</url-pattern>
	</servlet-mapping>
</web-app>

FirstServlet.java:-

import javax.servlet.*;
import javax.servlet.http.*;
import java.io.*;
public class FirstServlet extends HttpServlet
	{
	public void doGet(HttpServletRequest request,HttpServletResponse response)
					throws ServletException,IOException
	{
		String name=request.getParameter("txtName");
		response.setContentType("text/html");
		PrintWriter out=response.getWriter();
		out.println("Welcome,"+name);
		out.close();
	}
}
