Create an application for access custom TagLibrary into Servlet
Previous Home Next

In this Servlet application, custom TagLibrary accesses a Servlet application.Consider following steps for creating this application-

Step-1 Create a new web project and name it as TagServlet. Right click on WEB-INF folder and go to New -> other ->New File, window open (as show in below picture). Now choose Web from the Categories list and and Tag Library Descriptor from File Types list, then click on Next button.

Step-2 New Tag Library Descriptor Wizard window open, Specify the values as shown in below picture. After fill all the entry click on Next button.

Step-3 Go to source package and name it as r4r.TagLib.Servlet, now right click over package and go to New -> other ->New File, window open (as show into below picture). Now choose Web from the Categories list and and Tag Handler from File Types list, then click on Next button.

Step-4 New Tag Handler window open, Specify the values as shown in below picture. After fill all the entry click on Next button.

Step-5 Now add Tag Handler information along with attributes into TLD file.

  <servlet-name>TagServlet</servlet-name>
  <servlet-class>r4r.TagLib.Servlet.TagServlet</servlet-class>
 </servlet>
 <servlet-mapping>
  <servlet-name>TagServlet</servlet-name>
  <url-pattern>/TagServlet</url-pattern>
 </servlet-mapping>
 <session-config>
  <session-timeout>
    30
  </session-timeout>
 </session-config>
</web-app>

primeFactor.tld

<taglib version="2.1" 
  xmlns="http://java.sun.com/xml/ns/javaee"
  xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
  xsi:schemaLocation="http://java.sun.com/xml/ns/javaee 
    http://java.sun.com/xml/ns/javaee/web-jsptaglibrary_2_1.xsd">
 <tlib-version>1.0</tlib-version>
 <short-name>primefactor</short-name>
 <uri>/WEB-INF/tlds/primeFactor</uri>
 <tag>
  <name>FactorTagHandler</name>
  <tag-class>r4r.TagLib.Servlet.FactorTagHandler</tag-class>
  <body-content>empty</body-content> 
  <attribute>
   <name>number</name>
    <required>true</required>
    <rtexprvalue>true</rtexprvalue> 
    <type>double</type>
  </attribute>
 </tag>
</taglib>

Filter Program

/*
 * Save as a FactorTagHandler.java
 */
package r4r.TagLib.Servlet;
import java.io.PrintWriter;
import java.util.logging.Level;
import java.util.logging.Logger;
import javax.servlet.jsp.JspException;
import javax.servlet.jsp.JspWriter;
import javax.servlet.jsp.tagext.TagSupport;
/**
 *
 * @author R4R
 */
public class FactorTagHandler extends TagSupport {

 private double number;

 /** 
  * Creates new instance of tag handler 
  */
 public FactorTagHandler() {
  super();
 }

 @Override
 public int doStartTag() throws JspException {
  JspWriter out = pageContext.getOut();
  PrintWriter pw = new PrintWriter(out);
  try {
//Check input number is positive or negative
if (number >= 0) {
 //Check number is even or odd
 if ((number % 2.0) != 1) {
  pw.printf
	("Input number <b>(%.3f)</b> is positive and Even <br/>", number);
 } 
 else {
  pw.printf
	("Input number <b>(%.3f)</b> is positive and ODD <br/> ", number);
 }

 // prime factorization 
 pw.printf("The prime factorization of <b>(%.3f)</b> ==", number);
 for (double i = 2.0; i <= number / i; i++) {
  // Continue loop until number divide by i
  while (number % i == 0) {
pw.print(i + " x ");
number = number / i;
  }
 }
 pw.print(number);
} else if (number == 0) {
 pw.printf("Input number <b>(%.3f)</b> is Zero <br/>", number);
} else {
 pw.printf("Input number <b>(%.3f)</b> is negative <br/>", number);
}
  } finally {
try {
 out.flush(); //free resource
 out.close();
} catch (java.io.IOException ex) {
 Logger.getLogger(FactorTagHandler.class.getName()).log(Level.SEVERE,
"IOExcpetion generated while close streaming", ex);
}
  }
  return SKIP_BODY;
 }

 @Override
 public int doEndTag() throws JspException {
  return EVAL_PAGE;
 }

 //--  getter/setter of number
 public double getNumber() {
  return number;
 }

 public void setNumber(double number) {
  this.number = number;
 }
}
