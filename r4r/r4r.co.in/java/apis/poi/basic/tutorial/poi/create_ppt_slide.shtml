In this page of the tutorials we are going to create a PowerPoint slide in java application.

Example

package r4r;
import org.apache.poi.hslf.HSLFSlideShow;
import org.apache.poi.hslf.model.Slide;
import org.apache.poi.hslf.usermodel.SlideShow;
import java.io.*;
public class createpptslide {
	public static void main(String str[])
	  {
	  try{
	  SlideShow Show = new SlideShow();
	  Slide slide = Show.createSlide();
	  FileOutputStream fos = new FileOutputStream("slideppt.ppt");
	  Show.write(fos);
	  fos.close();
	  }catch(Exception e){}
	  }
}
