package r4r.Mobile.Basic;

import java.io.IOException;
import javax.microedition.lcdui.*;
import javax.microedition.midlet.*;

/**
 * @author R4R
 */
public class addImage extends MIDlet {

    private Form form;
    private Image image;

    public addImage() {
        form = new Form("Add Image into Form");
        try {
            // Path of image [ r4r.Mobile -> scr -> j2me_logo.png ]
            image = Image.createImage("/j2me_logo.png");
        } catch (IOException ex) {
            System.out.println("Problem reading with image" + ex);
        }
        form.append(image);
    }

    public void startApp() {
        Display.getDisplay(this).setCurrent(form);
    }

    public void pauseApp() {
    }

    public void destroyApp(boolean unconditional) {
    }
}
