How to Create Screen Saver using C#
Previous Home Next

In this Example I will show you how to make a screen saver by using c# windows form.

Step 1: Add a form in your application and make it form border style none. Set transparency key as form background color so it will make your form transparent.

Step 2: Put a timer control on your form and set it's interval as you want to display picture per millisecond on the computer screen.

Step 3: Write Code in .CS File

CS_Code

using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Text;
using System.Windows.Forms;
namespace screensever
{
public partial class Form1 : Form
{
public Form1()
{
InitializeComponent();   
}
private void Form1_Load(object sender, EventArgs e)
{  
timer1.Start();
}
int x, y;
private void timer1_Tick(object sender, EventArgs e)
{
x = Convert.ToInt32((Microsoft.VisualBasic.VBMath.Rnd())
	* (this.ClientSize.Width));
y = Convert.ToInt32((Microsoft.VisualBasic.VBMath.Rnd())
	* (this.ClientSize.Height)); 
screenStart();
}
private void screenStart()
{
PictureBox pbox = new PictureBox();
pbox.Location = new Point(x,y);
pbox.Height = 100;
pbox.Width = 100;
pbox.Image = global::screensever.Properties
	.Resources.thank_you_blue_rose;
pbox.SizeMode = System.Windows.Forms
	.PictureBoxSizeMode.StretchImage;
Controls.Add(pbox);  
}
private void Form1_KeyUp(object sender, KeyEventArgs e)
{
if (Keys.Escape == e.KeyCode)
{
this.Close();
}
} 
}
}
