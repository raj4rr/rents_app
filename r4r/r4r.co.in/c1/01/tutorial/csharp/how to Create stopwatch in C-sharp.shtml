private void timer1_Tick(object sender, System.EventArgs e)
{
TimeSpan span = DateTime.Now.Subtract(da);

this.label1.Text = span.Hours.ToString() + " : " + 
span.Minutes.ToString() + " : " 
+ span.Seconds.ToString() + " : " + 
span.Milliseconds.ToString();
}

}

}
}
