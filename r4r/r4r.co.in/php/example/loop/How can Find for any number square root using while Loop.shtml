<?php
$number = 41;
$g=1.0;
$pre= 0.0000001;
$g_squared = $g * $g;
while (($g_squared - $number > $pre) or ($g_squared - $number < - $pre))
{
  $g = ($g + ($number / $g)) / 2;
  $g_squared = $g * $g;
}
print("$g squared = $g_squared<BR>");
?>
