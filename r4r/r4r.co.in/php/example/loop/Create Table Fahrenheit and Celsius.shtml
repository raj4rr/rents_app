<?php
$Fahrenheit = -50;
$stop_Fahrenheit = 50;

print '<table>';
print '<tr><th>Fahrenheit</th><th>Celsius</th></tr>';
while ($Fahrenheit<= $stop_Fahrenheit) {
    $celsius = ($Fahrenheit - 32) * 5 / 9;
    print "<tr><td>$Fahrenheit</td><td>$celsius</td></tr>";
    $Fahrenheit+= 5;
}
print '</table>';
?>
