In PHP using the Comparison or Relational Operator :

OPERATOROPERATION
= = Equal
= = = Identical
!= Not equal
<> Not equal
!= = Not identical
< Less than
> Greater than
< = Less than or Equal to
>= Greater than or Equal to

	<?php
	$s = 10;

	printf ("$s == $a :%d <br>", $s==$a);

	printf ("$i > $o :%d <br>", $i > $o);
	printf ("$s < $a :%d <br>", $s < $a);
