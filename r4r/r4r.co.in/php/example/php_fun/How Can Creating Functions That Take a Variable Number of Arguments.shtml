<?php
  function addanything ()
{
    $t = 0;
    $a = func_get_args ();
    for ($i = 0; $i < count ($a); $i++)
{
      if (is_int ($a[$i])){
        $t += $a[$i];
      }
    }
    return $t;
  }
  echo addanything (100,150,120,130,140) . "<br />"; 
?>
