<h1>Get parameter count from func_num_args()</h1>
<?
function addNums() {
    $ret = "<table>";
    for ($x=0; $x<func_num_args (); $x++) {
        $arg = func_get_arg ($x);
        $result += $arg;
        $ret .= "<tr><td>number ". ($x+1).": 
		</td><td>$arg</td></tr>";
    }
    $ret .= "<tr><td>result: </td><td>$result</td><tr>";
    $ret .= "</table>";
    return $ret;
}

print addNums (19, 20, 34, 42, 75);
?>
