<?php
$a = array('A', 'B', 'C', 'D', 'E');
$admin = array('F', 'G');

function check_access($username, $adminonly = false) 
{
    global $a, $admin;

    if (in_array($username, $admin)) 
	{
        return true;
    }

    if (!($adminonly) && in_array($username, $a)) 
	{
        return true;
    }

    return false;
}
