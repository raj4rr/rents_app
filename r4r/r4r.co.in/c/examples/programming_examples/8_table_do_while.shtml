Table of any number using Do While loop 
#include<stdio.h>
#include<conio.h>
void main()
{
	int n, i, product;
	i=1;
	clrscr();
	printf("enter the number to make the table .");
	scanf("%d",&n);
	do
	{
		product=n*i;
		printf ("\n %d*%d=%d",n, i, product);
		i++;
	}
	while(i<=10);
	getch();
}
