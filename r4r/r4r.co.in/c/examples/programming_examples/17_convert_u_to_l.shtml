Program to convert upper case to lower case 
/** Convert upper case to lower case **/

#include<stdio.h>
#include<conio.h>
void main()
{
	char l[45];
	int i;
	clrscr();
	printf("Enter text in upper case");
	scanf("%s",l);
	printf("\n\n\n Uppercase text converted to lower case \n");
	for(i=0;l[i]!='\0';i++)
	{
		printf("%c",tolower(l[i]));
	}
	getch();
}
